# eaiku Plan 02: Web App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A local Next.js app at `localhost:3000` that shows the notes in `~/eaiku`: a notes list (glass cards grid, search, tag filter, series groups), a note page (article + sticky side rail), a landing page (CSS 3D "EAIKU", click to zoom out, launcher pills), and a Plugins page (health checks, enable toggle).

**Architecture:** `apps/web` is a Next.js App Router app. It reads SQLite **read-only** and never runs migrations (the `eaiku` CLI owns the database). Plugins own their UI: `plugins/catat/web.tsx` exports an `EaikuWebPlugin` whose `render()` loads data and returns React elements. One catch-all route, `app/p/[plugin]/[[...slug]]`, hands the request to the plugin. Styling is SCSS modules, one kebab-case folder per component, and global CSS custom properties for the dark glass theme.

**Tech Stack:** Next.js (App Router, React 19), SCSS modules (`sass`), `react-markdown`, Fontsource variable fonts (Bricolage Grotesque, Inter, JetBrains Mono), better-sqlite3 (read-only), Vitest.

**Spec:** `docs/superpowers/specs/2026-09-28-eaiku-v0.1-design.md` §8

**Design decisions made with the user (2026-09-28):**
- Note page: one readable article column + a sticky glass side rail ("On this page", back link, series prev/next). On mobile the rail becomes a sticky, horizontally scrolling row at the top.
- Notes list: glass cards grid with thumbnail, "Part x/y" badge, title, hook, and tags. Series parts are grouped under the video title. Search box and tag chips on top.
- Visual style: friendly, modern, dark, fun. Dark background with two soft color glows, glass panels, one bright accent (mint `#7df9c9`), rounded corners, soft motion.
- Build order: notes list and note page first, then the landing page, then the Plugins page.

**Code rules (from the user):**
- Styles live in SCSS module files, never inline `style` props. One kebab-case folder per component: `components/note-card/NoteCard.tsx` + `NoteCard.module.scss`.
- Inline arrow functions in JSX only when the body is one line. Otherwise extract a named function.

**Simplifications versus the spec (recorded in the spec in Task 9):**
- No Turborepo. The root `pnpm dev` script runs the web app. There is no build to cache yet.
- The plugin list is hand-written (`apps/web/lib/plugins.ts`), like the CLI's list. No generated registry.
- The enable toggle affects the web app only (pages and launcher pills). The CLI ignores it.

```ts
// For the web app: read-only, never creates the file, never runs migrations.
export function openDbReadOnly(home: string): Sqlite | null {
  const file = path.join(home, "eaiku.db");
  if (!fs.existsSync(file)) return null;
  return new Database(file, { readonly: true, fileMustExist: true });
}
```

`packages/core/src/config.ts`:

```ts
import fs from "node:fs/promises";
import path from "node:path";

// <home>/config.json. Only the web app reads it in v0.1.
export interface EaikuConfig {
  disabledPlugins: string[];
}

const FILE = "config.json";

export async function readConfig(home: string): Promise<EaikuConfig> {
  try {
    const raw = JSON.parse(await fs.readFile(path.join(home, FILE), "utf8")) as Partial<EaikuConfig>;
    const disabled = Array.isArray(raw.disabledPlugins) ? raw.disabledPlugins : [];
    return { disabledPlugins: disabled.filter((id): id is string => typeof id === "string") };
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === "ENOENT") return { disabledPlugins: [] };
    throw error;
  }
}

export async function setPluginEnabled(home: string, pluginId: string, enabled: boolean): Promise<EaikuConfig> {
  const disabled = new Set((await readConfig(home)).disabledPlugins);
  if (enabled) disabled.delete(pluginId);
  else disabled.add(pluginId);

  const next: EaikuConfig = { disabledPlugins: [...disabled].sort() };
  await fs.mkdir(home, { recursive: true });
  await fs.writeFile(path.join(home, FILE), JSON.stringify(next, null, 2) + "\n");
  return next;
}
```

Append to `packages/core/src/plugin.ts`:

```ts
export interface HealthResult {
  label: string;
  ok: boolean;
  hint?: string;
}

// Shared by `eaiku doctor` and the web Plugins page.
export async function runHealthChecks(
  plugin: Pick<EaikuPlugin, "healthChecks">,
  ctx: PluginContext,
): Promise<HealthResult[]> {
  return Promise.all(
    plugin.healthChecks.map(async (check) => {
      try {
        return { label: check.label, ...(await check.run(ctx)) };
      } catch (error) {
        return { label: check.label, ok: false, hint: (error as Error).message };
      }
    }),
  );
}
```

```ts
import { createContext, runHealthChecks, type EaikuPlugin, type HealthResult } from "@eaiku/core";

export async function runDoctor(plugins: EaikuPlugin[], home: string) {
  const checks: (HealthResult & { plugin: string })[] = [];
  for (const plugin of plugins) {
    const ctx = createContext(plugin, home);
    for (const result of await runHealthChecks(plugin, ctx)) checks.push({ plugin: plugin.id, ...result });
    ctx.db.close();
  }
  return { home, ok: checks.every((c) => c.ok), checks };
}
```

```ts
export interface NoteRow {
  id: string;
  title: string;
  videoId: string;
  videoTitle: string;
  channel: string | null;
  startSec: number;
  endSec: number;
  seriesId: string | null;
  part: number | null;
  total: number | null;
  hook: string;
  createdAt: string;
  tags: string[];
}

export interface SeriesSibling {
  id: string;
  part: number;
  title: string;
}

const TAG_SEPARATOR = "\u001f";

// Free text → FTS5 query. Every word becomes a quoted prefix term, so symbols can never break the syntax.
export function toFtsQuery(input: string): string | null {
  const words = input.toLowerCase().match(/[\p{L}\p{N}]+/gu) ?? [];
  return words.length > 0 ? words.map((w) => `"${w}"*`).join(" ") : null;
}

export function listNotes(sqlite: Sqlite, options: { q?: string; tag?: string } = {}): NoteRow[] {
  const match = options.q ? toFtsQuery(options.q) : null;
  const rows = sqlite
    .prepare(
      `SELECT n.id, n.title, n.video_id AS videoId, COALESCE(n.video_title, n.title) AS videoTitle,
              n.channel, n.start_sec AS startSec, n.end_sec AS endSec, n.series_id AS seriesId,
              n.part, n.total, n.hook, n.created_at AS createdAt,
              (SELECT group_concat(t.tag, char(31)) FROM catat_note_tags t WHERE t.note_id = n.id) AS tags
       FROM catat_notes n
       WHERE (@match IS NULL OR n.id IN (SELECT note_id FROM catat_notes_fts WHERE catat_notes_fts MATCH @match))
         AND (@tag IS NULL OR EXISTS (SELECT 1 FROM catat_note_tags t WHERE t.note_id = n.id AND t.tag = @tag))
       ORDER BY n.created_at DESC, n.series_id, n.part`,
    )
    .all({ match, tag: options.tag ?? null }) as (Omit<NoteRow, "tags"> & { tags: string | null })[];
  return rows.map((row) => ({ ...row, tags: row.tags ? row.tags.split(TAG_SEPARATOR).sort() : [] }));
}

export function listTags(sqlite: Sqlite): { tag: string; count: number }[] {
  return sqlite
    .prepare("SELECT tag, count(*) AS count FROM catat_note_tags GROUP BY tag ORDER BY count DESC, tag")
    .all() as { tag: string; count: number }[];
}

export function getSeriesNotes(sqlite: Sqlite, seriesId: string): SeriesSibling[] {
  return sqlite
    .prepare("SELECT id, part, title FROM catat_notes WHERE series_id = ? AND part IS NOT NULL ORDER BY part")
    .all(seriesId) as SeriesSibling[];
}
```

```ts
import type { NoteRow } from "../db/queries";

export interface SeriesGroup {
  seriesId: string;
  videoTitle: string;
  channel: string | null;
  total: number;
  notes: NoteRow[];
}

// Keeps the input order (newest first) for groups and singles, and sorts parts inside a group.
export function groupNotes(rows: NoteRow[]): { series: SeriesGroup[]; singles: NoteRow[] } {
  const series = new Map<string, SeriesGroup>();
  const singles: NoteRow[] = [];

  for (const row of rows) {
    if (!row.seriesId) {
      singles.push(row);
      continue;
    }
    let group = series.get(row.seriesId);
    if (!group) {
      group = { seriesId: row.seriesId, videoTitle: row.videoTitle, channel: row.channel, total: row.total ?? 0, notes: [] };
      series.set(row.seriesId, group);
    }
    group.notes.push(row);
  }
  for (const group of series.values()) group.notes.sort((a, b) => (a.part ?? 0) - (b.part ?? 0));
  return { series: [...series.values()], singles };
}
```

```tsx
import { openDbReadOnly, readConfig, resolveHome } from "@eaiku/core";
import { webPlugins } from "@/lib/plugins";
import { firstValues } from "@/lib/search-params";

// Notes change on disk at any time, so never prerender.
export const dynamic = "force-dynamic";

interface Props {
  params: Promise<{ plugin: string; slug?: string[] }>;
  searchParams: Promise<Record<string, string | string[] | undefined>>;
}

export default async function PluginRoute({ params, searchParams }: Props) {
  const { plugin: pluginId, slug = [] } = await params;
  const home = resolveHome();
  const config = await readConfig(home);
  const plugin = webPlugins.find((p) => p.id === pluginId);
  if (!plugin || config.disabledPlugins.includes(pluginId)) notFound();

  const db = openDbReadOnly(home);
  try {
    const content = await plugin.render({ slug, searchParams: firstValues(await searchParams), home, db });
    if (content === null) notFound();
    return content;
  } finally {
    db?.close();
  }
}
```

```tsx
"use client";

import { useEffect, useState, type ReactNode } from "react";
import styles from "./SectionRail.module.scss";

export interface RailItem {
  id: string;
  label: string;
}

// Highlights the section that is at the top of the screen.
export function SectionRail({ items, children }: { items: RailItem[]; children?: ReactNode }) {
  const [activeId, setActiveId] = useState(items[0]?.id);

  useEffect(() => {
    const handleEntries = (entries: IntersectionObserverEntry[]) => {
      const visible = entries.find((entry) => entry.isIntersecting);
      if (visible) setActiveId(visible.target.id);
    };
    const observer = new IntersectionObserver(handleEntries, { rootMargin: "0px 0px -70% 0px" });
    for (const item of items) {
      const element = document.getElementById(item.id);
      if (element) observer.observe(element);
    }
    return () => observer.disconnect();
  }, [items]);

  return (
    <nav className={`glass ${styles.rail}`} aria-label="On this page">
      <p className={styles.label}>On this page</p>
      <ol className={styles.list}>
        {items.map((item) => (
          <li key={item.id}>
            <a href={`#${item.id}`} className={item.id === activeId ? `${styles.item} ${styles.active}` : styles.item}>
              {item.label}
            </a>
          </li>
        ))}
      </ol>
      {children ? <div className={styles.extra}>{children}</div> : null}
    </nav>
  );
}
```

```tsx
import { useState, type PointerEvent } from "react";
import type { LauncherAction } from "@eaiku/core";
import { LauncherPill } from "../launcher-pill/LauncherPill";
import styles from "./Landing.module.scss";

export type LandingAction = LauncherAction & { pluginId: string };

const MAX_TILT_X = 14;
const MAX_TILT_Y = 22;

export function Landing({ actions }: { actions: LandingAction[] }) {
  const [awake, setAwake] = useState(false);
  const [toast, setToast] = useState<string | null>(null);

  // Tilt the word toward the cursor with CSS variables (no inline style props).
  const handlePointerMove = (event: PointerEvent<HTMLElement>) => {
    const rect = event.currentTarget.getBoundingClientRect();
    const x = (event.clientX - rect.left) / rect.width - 0.5;
    const y = (event.clientY - rect.top) / rect.height - 0.5;
    event.currentTarget.style.setProperty("--tilt-x", `${(-y * MAX_TILT_X * 2).toFixed(2)}deg`);
    event.currentTarget.style.setProperty("--tilt-y", `${(x * MAX_TILT_Y * 2).toFixed(2)}deg`);
  };

  const handleCopy = async (command: string) => {
    await navigator.clipboard.writeText(command);
    setToast(`Copied “${command.trim()}”. Paste it in Claude Code with a YouTube link.`);
    window.setTimeout(() => setToast(null), 3000);
  };

  return (
    <main className={awake ? `${styles.stage} ${styles.awake}` : styles.stage} onPointerMove={handlePointerMove}>
      <button type="button" className={styles.wordmark} onClick={() => setAwake((value) => !value)} aria-expanded={awake}>
        <span className={styles.word}>EAIKU</span>
      </button>

      <p className={styles.hint}>{awake ? "What shall we do?" : "Click to wake me up"}</p>

      <nav className={styles.actions} aria-hidden={!awake}>
        {actions.map((action) => (
          <LauncherPill key={`${action.pluginId}-${action.label}`} action={action} onCopy={handleCopy} hidden={!awake} />
        ))}
      </nav>

      {toast ? (
        <p className={`glass ${styles.toast}`} role="status">
          {toast}
        </p>
      ) : null}

      <footer className={styles.footer}>© 2026 eaiku · local-first · your agent does the work</footer>
    </main>
  );
}
```

```ts
"use server";

import { revalidatePath } from "next/cache";
import { resolveHome, setPluginEnabled } from "@eaiku/core";
import { plugins } from "@/lib/plugins";

export async function togglePlugin(formData: FormData): Promise<void> {
  const id = String(formData.get("id"));
  const enabled = formData.get("enabled") === "true";
  if (!plugins.some((plugin) => plugin.id === id)) return;
  await setPluginEnabled(resolveHome(), id, enabled);
  revalidatePath("/", "layout");
}
```

`apps/web/app/plugins/page.tsx`:

```tsx
import { createStorage, openDbReadOnly, readConfig, resolveHome, runHealthChecks, type HealthResult } from "@eaiku/core";
import { PluginRow } from "@/components/plugin-row/PluginRow";
import { plugins } from "@/lib/plugins";
import styles from "./page.module.scss";

export const dynamic = "force-dynamic";

const NO_DATABASE: HealthResult[] = [
  { label: "Database", ok: false, hint: "No database yet. Run `eaiku doctor` once." },
];

export default async function PluginsPage() {
  const home = resolveHome();
  const { disabledPlugins } = await readConfig(home);
  const db = openDbReadOnly(home);

  try {
    const rows = await Promise.all(
      plugins.map(async (plugin) => ({
        plugin,
        enabled: !disabledPlugins.includes(plugin.id),
        checks: db ? await runHealthChecks(plugin, { home, db, storage: createStorage(home, plugin.id) }) : NO_DATABASE,
      })),
    );

    return (
      <main className="page-main">
        <h1 className={styles.heading}>Plugins</h1>
        <p className={styles.intro}>
          Each plugin gives your agent a new skill and this app a new page. Data lives in <code>{home}</code>.
        </p>
        <div className={styles.list}>
          {rows.map(({ plugin, enabled, checks }) => (
            <PluginRow
              key={plugin.id}
              id={plugin.id}
              name={plugin.name}
              description={plugin.description}
              version={plugin.version}
              enabled={enabled}
              checks={checks}
            />
          ))}
        </div>
      </main>
    );
  } finally {
    db?.close();
  }
}
```
